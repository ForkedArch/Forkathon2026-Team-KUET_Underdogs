-- STEP 2: ENUM TYPES

do $$
begin
    if not exists (
        select 1 from pg_type where typname = 'user_role'
    ) then
        create type public.user_role as enum (
            'student',
            'manager',
            'admin'
        );
    end if;

    if not exists (
        select 1 from pg_type where typname = 'booking_status'
    ) then
        create type public.booking_status as enum (
            'pending',
            'approved',
            'rejected',
            'cancelled',
            'completed',
            'no_show'
        );
    end if;

    if not exists (
        select 1 from pg_type where typname = 'space_status'
    ) then
        create type public.space_status as enum (
            'available',
            'maintenance',
            'restricted'
        );
    end if;
end
$$;

-- ============================================================
-- STEP 4: SPACES TABLE
-- ============================================================

create table if not exists public.spaces (
    id uuid primary key default gen_random_uuid(),

    name text not null,

    building text not null,

    floor text,

    capacity integer not null default 1
        check (capacity > 0),

    description text,

    purpose text,

    projector boolean not null default false,

    ac boolean not null default false,

    computers boolean not null default false,

    microphone boolean not null default false,

    permission text not null default 'All Students',

    status public.space_status not null default 'available',

    maintenance_until timestamptz,

    created_at timestamptz not null default now()
);


-- ============================================================
-- STEP 5: BOOKINGS TABLE
-- ============================================================

create table if not exists public.bookings (
    id uuid primary key default gen_random_uuid(),

    space_id uuid not null
        references public.spaces(id)
        on delete cascade,

    user_id uuid not null
        references public.profiles(id)
        on delete cascade,

    purpose text not null,

    participant_count integer not null
        check (participant_count > 0),

    start_at timestamptz not null,

    end_at timestamptz not null,

    status public.booking_status not null default 'pending',

    checked_in boolean not null default false,

    created_at timestamptz not null default now(),

    constraint valid_booking_time
        check (end_at > start_at),

    constraint valid_participant_count
        check (participant_count > 0)
);


-- ============================================================
-- STEP 6: USAGE LOGS TABLE
-- ============================================================

create table if not exists public.usage_logs (
    id uuid primary key default gen_random_uuid(),

    booking_id uuid not null unique
        references public.bookings(id)
        on delete cascade,

    user_id uuid not null
        references public.profiles(id)
        on delete cascade,

    space_id uuid not null
        references public.spaces(id)
        on delete cascade,

    check_in timestamptz,

    check_out timestamptz,

    no_show boolean not null default false,

    created_at timestamptz not null default now(),

    constraint valid_usage_time
        check (
            check_out is null
            or check_in is null
            or check_out >= check_in
        )
);

create or replace function public.handle_new_user()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin

    insert into public.profiles (
        id,
        name,
        student_id,
        department,
        mobile,
        role
    )
    values (
        new.id,
        coalesce(
            new.raw_user_meta_data ->> 'name',
            'New User'
        ),
        new.raw_user_meta_data ->> 'student_id',
        new.raw_user_meta_data ->> 'department',
        new.raw_user_meta_data ->> 'mobile',
        'student'
    );

    return new;

end;
$$;


drop trigger if exists on_auth_user_created
on auth.users;


create trigger on_auth_user_created
after insert on auth.users
for each row
execute function public.handle_new_user();

-- STEP 15: BOOKING CONFLICT CHECK FUNCTION
-- ============================================================

create or replace function public.has_booking_conflict(
    p_space_id uuid,
    p_start_at timestamptz,
    p_end_at timestamptz,
    p_exclude_booking_id uuid default null
)
returns boolean
language sql
stable
security definer
set search_path = public
as $$
    select exists (
        select 1
        from public.bookings b
        where b.space_id = p_space_id

        and b.status = 'approved'

        and (
            p_exclude_booking_id is null
            or b.id <> p_exclude_booking_id
        )

        and b.start_at < p_end_at
        and b.end_at > p_start_at
    );
$$;


-- ============================================================
-- STEP 16: APPROVE BOOKING FUNCTION
-- ============================================================

create or replace function public.approve_booking(
    p_booking_id uuid
)
returns boolean
language plpgsql
security definer
set search_path = public
as $$
declare

    selected_booking public.bookings%rowtype;

    space_record public.spaces%rowtype;

begin

    -- Check manager permission

    if not public.is_manager() then
        raise exception 'Only managers can approve bookings';
    end if;


    -- Get booking

    select *
    into selected_booking
    from public.bookings
    where id = p_booking_id
    for update;


    if selected_booking.id is null then
        raise exception 'Booking not found';
    end if;


    if selected_booking.status <> 'pending' then
        raise exception 'Only pending bookings can be approved';
    end if;


    -- Get space

    select *
    into space_record
    from public.spaces
    where id = selected_booking.space_id;


    if space_record.id is null then
        raise exception 'Space not found';
    end if;


    -- Check whether space is usable

    if space_record.status <> 'available' then
        raise exception 'Space is not currently usable';
    end if;


    -- Check capacity

    if selected_booking.participant_count > space_record.capacity then
        raise exception 'Participant count exceeds room capacity';
    end if;


    -- Check booking conflict

    if public.has_booking_conflict(
        selected_booking.space_id,
        selected_booking.start_at,
        selected_booking.end_at,
        selected_booking.id
    ) then

        raise exception 'This space is already booked during this time';

    end if;


    -- Approve booking

    update public.bookings

    set status = 'approved'

    where id = selected_booking.id;


    return true;

end;
$$;

    p_booking_id uuid

security definer

as $$
begin

    if not public.is_manager() then
        raise exception 'Only managers can reject bookings';
    end if;


    update public.bookings

    set status = 'rejected'

    where id = p_booking_id
    and status = 'pending';


    if not found then
        raise exception 'Pending booking not found';
    end if;


    return true;

end;
$$;


-- ============================================================
-- STEP 18: CHECK-IN FUNCTION
-- ============================================================

create or replace function public.check_in_booking(
    p_booking_id uuid
)
returns boolean
language plpgsql
security definer
set search_path = public
as $$
declare

    selected_booking public.bookings%rowtype;

begin

    select *
    into selected_booking

    from public.bookings

    where id = p_booking_id

    and user_id = auth.uid()

    for update;


    if selected_booking.id is null then
        raise exception 'Booking not found';
    end if;


    if selected_booking.status <> 'approved' then
        raise exception 'Only approved bookings can be checked in';
    end if;


    if selected_booking.checked_in then
        raise exception 'Already checked in';
    end if;


    -- Mark booking as checked in

    update public.bookings

    set checked_in = true

    where id = p_booking_id;


    -- Create usage log

    insert into public.usage_logs (
        booking_id,
        user_id,
        space_id,
        check_in
    )

    values (
        selected_booking.id,
        selected_booking.user_id,
        selected_booking.space_id,
        now()
    )

    on conflict (booking_id)

    do update set
        check_in = now(),
        no_show = false;


    return true;

end;
$$;


-- ============================================================
-- STEP 19: CHECKOUT FUNCTION
-- ============================================================

create or replace function public.check_out_booking(
    p_booking_id uuid
)
returns boolean
language plpgsql
security definer
set search_path = public
as $$
declare

    selected_booking public.bookings%rowtype;

begin

    select *
    into selected_booking

    from public.bookings

    where id = p_booking_id

    and user_id = auth.uid();


    if selected_booking.id is null then
        raise exception 'Booking not found';
    end if;


    if not selected_booking.checked_in then
        raise exception 'You have not checked in';
    end if;


    update public.usage_logs

    set check_out = now()

    where booking_id = p_booking_id;


    update public.bookings

    set status = 'completed'

    where id = p_booking_id;


    return true;

end;
$$;

        user_id,
        space_id,
        no_show
    )

    values (
        selected_booking.id,
        selected_booking.user_id,
        selected_booking.space_id,
        true
    )

    on conflict (booking_id)

    do update set
        no_show = true;


    return true;

end;
$$;

## ❔ Problem

### Problem Statement

The Empty Room

A classroom is empty for three hours every afternoon. Somewhere else, a group of students is desperately looking for a place to work. A laboratory is available, but nobody knows whether they are allowed to use it. A meeting room has been reserved but nobody actually shows up.

There is space everywhere—but somehow, nobody can find it when they need it.

What an absolute mess!
KUET needs this problem to be solutioned soon.

Maybe a digital solution for this that will be able to solve this?
The system could deal with classrooms, study spaces, meeting rooms, labs, sports facilities, community spaces, or even resources outside a campus.
Probably a system where you can know the availability, booking, conflicts, cancellations, usage, permissions, and fairness.

Brainstorming twist: Don't assume that "available" simply means "empty." What makes a space genuinely usable?


### 🤔 KUET_Underdogs's Understanding

Explain the problem in your own words. You may say a story mentioning your team member names!

Problems : 

1. No clear way to find spaces: Students do not know which rooms they could use.
2. Unknown availability: Students cannot check when a room is free or for how long.
3. Unclear permissions: Students do not know who can use a room or whether approval is required.
4. No clear booking process: Students must ask around or visit the department office.
5. Conflicting bookings: Multiple groups could request the same space at the same time.
6. Unused reservations: A group books a room but never shows up, blocking others.
7. Uncommunicated cancellations: Others do not know when a cancelled booking makes a room available again.
8. Missing room information: Students cannot check capacity, equipment, or suitability for their activity.
9. Unclear access instructions: A room may be free, but students do not know where to collect the key or whom to contact.
10. Unfair booking access: A few users could repeatedly reserve popular spaces. 
11. No usage history: Administrators lack a clear record of bookings, cancellations, and reported usage.
